269: // ── Hierarchical layout: BFS left→right ─────────────────────────────────────
270: function computeLayout(
271:   rawNodes: any[],
272:   rawEdges: any[],
273:   suspectAddress: string,
274: ): { nodes: Node[]; edges: Edge[] } {
275:   const H_GAP = 340; // ample horizontal space to prevent line clustering
276:   const V_GAP = 120; // ample vertical space between wallets
277: 
278:   // Compute incoming and outgoing transfers per node for instant card display
279:   const nodeTransfers = new Map<string, { received: number; sent: number; token: string; primaryTxAmount: number; inCount: number; outCount: number }>();
280:   for (const e of rawEdges) {
281:     const amt = parseFloat(e.amount || '0') || 0;
282:     const token = e.token || e.asset || 'ETH';
283:     const src = (e.source || '').toLowerCase();
284:     const tgt = (e.target || '').toLowerCase();
285: 
286:     if (!nodeTransfers.has(src)) {
287:       nodeTransfers.set(src, { received: 0, sent: 0, token, primaryTxAmount: amt, inCount: 0, outCount: 0 });
288:     }
289:     const s = nodeTransfers.get(src)!;
290:     s.sent += amt;
291:     s.outCount += 1;
292:     if (!s.token) s.token = token;
293: 
294:     if (!nodeTransfers.has(tgt)) {
295:       nodeTransfers.set(tgt, { received: 0, sent: 0, token, primaryTxAmount: amt, inCount: 0, outCount: 0 });
296:     }
297:     const t = nodeTransfers.get(tgt)!;
298:     t.received += amt;
299:     t.inCount += 1;
300:     t.primaryTxAmount = amt;
301:     if (!t.token) t.token = token;
302:   }
303: 
304:   // BFS to assign columns (depths)
305:   const depthMap = new Map<string, number>();
306:   const queue: string[] = [suspectAddress.toLowerCase()];
307:   depthMap.set(suspectAddress.toLowerCase(), 0);
308: 
309:   const adjOut = new Map<string, Set<string>>();
310:   const adjIn  = new Map<string, Set<string>>();
311:   for (const e of rawEdges) {
312:     const src = (e.source || '').toLowerCase();
313:     const tgt = (e.target || '').toLowerCase();
314:     if (!adjOut.has(src)) adjOut.set(src, new Set());
315:     if (!adjIn.has(tgt))  adjIn.set(tgt, new Set());
316:     adjOut.get(src)!.add(tgt);
317:     adjIn.get(tgt)!.add(src);
318:   }
319: 
320:   // Forward BFS
321:   while (queue.length) {
322:     const cur = queue.shift()!;
323:     const curDepth = depthMap.get(cur) ?? 0;
324:     for (const nxt of adjOut.get(cur) ?? []) {
325:       if (!depthMap.has(nxt)) {
326:         depthMap.set(nxt, curDepth + 1);
327:         queue.push(nxt);
328:       }
329:     }
330:   }
331:   // Backward BFS for nodes that feed INTO the suspect
332:   const backQueue: string[] = [suspectAddress.toLowerCase()];
333:   const backVisited = new Set([suspectAddress.toLowerCase()]);
334:   while (backQueue.length) {
335:     const cur = backQueue.shift()!;
336:     const curDepth = depthMap.get(cur) ?? 0;
337:     for (const prev of adjIn.get(cur) ?? []) {
338:       if (!depthMap.has(prev)) {
339:         depthMap.set(prev, curDepth - 1);
340:         if (!backVisited.has(prev)) { backQueue.push(prev); backVisited.add(prev); }
341:       }
342:     }
343:   }
344: 
345:   // Group nodes by depth column
346:   const columns = new Map<number, string[]>();
347:   for (const n of rawNodes) {
348:     const id = (n.id || '').toLowerCase();
349:     const d = depthMap.get(id) ?? 0;
350:     if (!columns.has(d)) columns.set(d, []);
351:     columns.get(d)!.push(n.id);
352:   }
353:   // Also add nodes not reachable from suspect (orphans → depth 999)
354:   for (const n of rawNodes) {
355:     const id = (n.id || '').toLowerCase();
356:     if (!depthMap.has(id)) {
357:       depthMap.set(id, 999);
358:       if (!columns.has(999)) columns.set(999, []);
359:       columns.get(999)!.push(n.id);
360:     }
361:   }
362: 
363:   // Assign pixel positions
364:   const posMap = new Map<string, { x: number; y: number }>();
365:   const sortedDepths = Array.from(columns.keys()).sort((a, b) => a - b);
366:   sortedDepths.forEach((depth, colIdx) => {
367:     const col = columns.get(depth)!;
368:     col.forEach((id, rowIdx) => {
369:       posMap.set(id, {
370:         x: colIdx * H_GAP,
371:         y: rowIdx * V_GAP - ((col.length - 1) * V_GAP) / 2,
372:       });
373:     });
374:   });
375: 
376:   const nodes: Node[] = rawNodes.map((n: any) => {
377:     const id = (n.id || '').toLowerCase();
378:     const tr = nodeTransfers.get(id);
379:     const isSuspect = id === suspectAddress.toLowerCase();
380: 
381:     return {
382:       id: n.id,
383:       type: 'walletNode',
384:       position: posMap.get(n.id) ?? { x: 0, y: 0 },
385:       data: {
386:         label: n.label || `${n.id.slice(0, 6)}...${n.id.slice(-4)}`,
387:         address: n.address || n.id,
388:         nodeType: isSuspect ? 'suspect' : (n.type || 'wallet'),
389:         balance: n.balance,
390:         riskScore: n.riskScore,
391:         riskLevel: n.riskLevel,
392:         entityLabel: n.entityLabel,
393:         dataSource: n.dataSource,
394:         receivedAmount: tr?.received ?? 0,
395:         sentAmount: tr?.sent ?? 0,
396:         primaryTxAmount: tr?.primaryTxAmount ?? 0,
397:         token: tr?.token || 'ETH',
398:         inCount: tr?.inCount ?? 0,
399:         outCount: tr?.outCount ?? 0,
400:         isSuspect,
401:       },
402:     };
403:   });
404: 
405:   const edges: Edge[] = rawEdges.map((e: any, i: number) => {
406:     const token = e.token || e.asset || 'ETH';
407:     const amount = e.amount ? parseFloat(e.amount) : 0;
408:     const isPending = e.status === 'PENDING';
409: 
410:     return {
411:       id: `e-${i}-${e.txHash || i}`,
412:       source: e.source,
413:       target: e.target,
414:       type: 'transferEdge',
415:       animated: isPending, // ONLY animate pending transactions to prevent visual barcode noise
416:       markerEnd: {
417:         type: MarkerType.ArrowClosed,
418:         color: isPending ? '#fbbf24' : '#475569',
419:       },
420:       data: {
421:         ...e,
422:         amount,
423:         token,
424:         status: e.status || 'CONFIRMED',
425:         edgeIndex: i,
426:       },
427:     };
428:   });
429: 
430:   return { nodes, edges };
431: }
